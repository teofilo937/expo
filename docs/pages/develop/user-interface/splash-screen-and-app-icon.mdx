// Protótipo de App de Contagem de Calorias com Abas - Adaptado para React Native
// Framework: React Native (usando estilo inline para compatibilidade com Expo)

import React, { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, ScrollView, StyleSheet } from 'react-native';

const tabsArray = ["Perfil", "Refeições", "Plano Alimentar", "Passos", "Reconhecimento de Foto"];

export default function CaloriaApp() {
  const [tabAtiva, setTabAtiva] = useState("Perfil");
  const [peso, setPeso] = useState('');
  const [idade, setIdade] = useState('');
  const [altura, setAltura] = useState('');
  const [descricaoRefeicao, setDescricaoRefeicao] = useState('');
  const [resultado, setResultado] = useState(null);

  const analisarDescricao = () => {
    setResultado({
      calorias: 540,
      proteinas: 20,
      carboidratos: 65,
      gorduras: 22,
    });
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.header}>Contador de Calorias</Text>

      {/* Navegação por Abas */}
      <View style={styles.tabBar}>
        {tabsArray.map((tab) => (
          <TouchableOpacity key={tab} onPress={() => setTabAtiva(tab)}>
            <Text style={[styles.tab, tabAtiva === tab && styles.activeTab]}>{tab}</Text>
          </TouchableOpacity>
        ))}
      </View>

      {/* Conteúdo da Aba Ativa */}
      {tabAtiva === "Perfil" && (
        <View style={styles.card}>
          <Text style={styles.title}>👋 Seja bem-vindo ao seu assistente de saúde!</Text>
          <Text style={styles.subtitle}>Preencha os seus dados para começarmos a personalizar sua experiência.</Text>
          <TextInput style={styles.input} placeholder="Peso (kg)" value={peso} onChangeText={setPeso} />
          <TextInput style={styles.input} placeholder="Idade" value={idade} onChangeText={setIdade} />
          <TextInput style={styles.input} placeholder="Altura (cm)" value={altura} onChangeText={setAltura} />
        </View>
      )}

      {tabAtiva === "Refeições" && (
        <View style={styles.card}>
          <Text style={styles.title}>🍽️ Análise Nutricional</Text>
          <Text style={styles.subtitle}>Descreva o que você comeu para obter os valores nutricionais.</Text>
          <TextInput
            style={[styles.input, { height: 100 }]}
            multiline
            placeholder="Ex: 100g de arroz, 1 ovo cozido"
            value={descricaoRefeicao}
            onChangeText={setDescricaoRefeicao}
          />
          <TouchableOpacity style={styles.button} onPress={analisarDescricao}>
            <Text style={styles.buttonText}>Analisar refeição</Text>
          </TouchableOpacity>

          {resultado && (
            <View style={styles.resultado}>
              <Text style={styles.resultTitle}>Resultado Nutricional</Text>
              <Text>Calorias: {resultado.calorias} kcal</Text>
              <Text>Proteínas: {resultado.proteinas} g</Text>
              <Text>Carboidratos: {resultado.carboidratos} g</Text>
              <Text>Gorduras: {resultado.gorduras} g</Text>
            </View>
          )}
        </View>
      )}

      {tabAtiva === "Plano Alimentar" && (
        <View style={styles.card}>
          <Text style={styles.title}>📅 Plano Alimentar</Text>
          <Text style={styles.subtitle}>Em breve você poderá gerar um plano alimentar personalizado com base no seu perfil.</Text>
        </View>
      )}

      {tabAtiva === "Passos" && (
        <View style={styles.card}>
          <Text style={styles.title}>🚶 Contador de Passos</Text>
          <Text style={styles.subtitle}>Futuramente, o app integrará com sensores para contar os seus passos diários.</Text>
        </View>
      )}

      {tabAtiva === "Reconhecimento de Foto" && (
        <View style={styles.card}>
          <Text style={styles.title}>📷 Reconhecimento de Refeição</Text>
          <Text style={styles.subtitle}>Tire ou envie uma foto da sua refeição e receba a estimativa de calorias automaticamente.</Text>
        </View>
      )}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: {
    padding: 16,
    backgroundColor: '#fff',
  },
  header: {
    fontSize: 24,
    fontWeight: 'bold',
    textAlign: 'center',
    marginBottom: 16,
  },
  tabBar: {
    flexDirection: 'row',
    justifyContent: 'space-around',
    marginBottom: 16,
  },
  tab: {
    paddingVertical: 8,
    fontSize: 16,
    color: 'gray',
  },
  activeTab: {
    color: '#007bff',
    fontWeight: 'bold',
    borderBottomWidth: 2,
    borderColor: '#007bff',
  },
  card: {
    marginBottom: 24,
    padding: 16,
    backgroundColor: '#f9f9f9',
    borderRadius: 8,
    shadowColor: '#000',
    shadowOffset: { width: 0, height: 2 },
    shadowOpacity: 0.1,
    shadowRadius: 4,
    elevation: 3,
  },
  title: {
    fontSize: 18,
    fontWeight: '600',
    marginBottom: 8,
  },
  subtitle: {
    fontSize: 14,
    color: '#666',
    marginBottom: 12,
  },
  input: {
    borderWidth: 1,
    borderColor: '#ccc',
    borderRadius: 6,
    padding: 10,
    marginBottom: 12,
  },
  button: {
    backgroundColor: '#007bff',
    padding: 12,
    borderRadius: 6,
    alignItems: 'center',
  },
  buttonText: {
    color: '#fff',
    fontWeight: '600',
  },
  resultado: {
    marginTop: 16,
  },
  resultTitle: {
    fontWeight: 'bold',
    marginBottom: 4,
  },
});
